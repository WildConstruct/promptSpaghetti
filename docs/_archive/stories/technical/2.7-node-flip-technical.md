# Technical Implementation Story 2.7: Node Flip Metadata Display

## Technical Overview

Implement a 3D flip animation system for React Flow nodes that reveals metadata on the back face, with support for keyboard/mouse/touch interactions, performance optimization, and accessibility.

## Architecture Components

### 1. Flippable Node Wrapper Component

```typescript
// packages/core/components/nodes/FlippableNode.tsx
import { useRef, useState, useEffect, useCallback } from 'react';
import { NodeProps, useReactFlow } from 'reactflow';
import { useLongPress } from '@/hooks/useLongPress';
import { useMetadataStore } from '@/stores/metadataStore';
import { useFlipManager } from '@/hooks/useFlipManager';

interface FlippableNodeProps extends NodeProps {
  nodeComponent: React.ComponentType<NodeProps>;
  enableFlip?: boolean;
}

export const FlippableNode: React.FC<FlippableNodeProps> = ({
  nodeComponent: NodeComponent,
  enableFlip = true,
  ...nodeProps
}) => {
  const { id, data, selected } = nodeProps;
  const [isFlipped, setIsFlipped] = useState(false);
  const [metadata, setMetadata] = useState<NodeMetadata | null>(null);
  const [isLoading, setIsLoading] = useState(false);

  const nodeRef = useRef<HTMLDivElement>(null);
  const flipManager = useFlipManager();
  const { getNodeMetadata, prefetchMetadata } = useMetadataStore();
  const { getViewport, fitView } = useReactFlow();

  // Prefetch metadata when node becomes visible
  useEffect(() => {
    if (enableFlip && isNodeInViewport()) {
      prefetchMetadata(id);
    }
  }, [id, enableFlip]);

  // Handle Alt+Click
  const handleClick = useCallback((e: React.MouseEvent) => {
    if (e.altKey && !e.target.closest('.node-controls')) {
      e.preventDefault();
      e.stopPropagation();
      flipNode();
    }
  }, []);

  // Handle long press for mobile
  const handleLongPress = useLongPress(flipNode, {
    threshold: 500,
    cancelOnMovement: 3,
    isDisabled: !enableFlip || isPanning()
  });

  // Handle keyboard navigation
  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
    if (e.key === 'Enter' && selected) {
      flipNode();
    } else if (e.key === 'Escape' && isFlipped) {
      setIsFlipped(false);
      flipManager.unregisterFlip(id);
    }
  }, [isFlipped, selected]);

  const flipNode = async () => {
    if (!enableFlip) return;

    // Check if flip would cause clipping
    if (wouldClipViewport()) {
      await adjustViewportForFlip();
    }

    // Register with flip manager for batch handling
    if (!isFlipped && flipManager.shouldBatch()) {
      flipManager.queueFlip(id, () => performFlip());
    } else {
      performFlip();
    }
  };

  const performFlip = async () => {
    if (!isFlipped && !metadata) {
      setIsLoading(true);
      const data = await getNodeMetadata(id);
      setMetadata(data);
      setIsLoading(false);
    }

    setIsFlipped(!isFlipped);
    flipManager.registerFlip(id, !isFlipped);
  };

  return (
    <div
      ref={nodeRef}
      className={`flippable-node-container ${isFlipped ? 'is-flipped' : ''}`}
      onClick={handleClick}
      onKeyDown={handleKeyDown}
      {...handleLongPress}
      role="group"
      aria-expanded={isFlipped}
      aria-label={isFlipped ? 'Metadata view' : 'Node view'}
      data-node-id={id}
    >
      <div className="flippable-node">
        <div className="node-front">
          <NodeComponent {...nodeProps} />
          {selected && enableFlip && (
            <FlipButton onClick={flipNode} aria-label="Flip to metadata" />
          )}
        </div>
        <div className="node-back">
          {isLoading ? (
            <MetadataSkeleton />
          ) : (
            <MetadataDisplay
              metadata={metadata}
              nodeId={id}
              onClose={() => setIsFlipped(false)}
            />
          )}
        </div>
      </div>
    </div>
  );
};
```

### 3. Flip Manager for Batch Processing

```typescript
// packages/core/services/FlipManager.ts
export class FlipManager {
  private flipQueue: Map<string, () => void> = new Map();
  private activeFlips: Set<string> = new Set();
  private animationFrame: number | null = null;
  private maxConcurrent = 10;

  queueFlip(nodeId: string, flipFn: () => void) {
    this.flipQueue.set(nodeId, flipFn);

    if (this.activeFlips.size < this.maxConcurrent) {
      this.processQueue();
    }
  }

  private processQueue() {
    if (this.animationFrame) return;

    this.animationFrame = requestAnimationFrame(() => {
      const batch = Array.from(this.flipQueue.entries()).slice(
        0,
        this.maxConcurrent - this.activeFlips.size
      );

      batch.forEach(([nodeId, flipFn]) => {
        this.flipQueue.delete(nodeId);
        this.activeFlips.add(nodeId);

        // Execute flip with stagger
        setTimeout(() => {
          flipFn();

          // Clean up after animation
          setTimeout(() => {
            this.activeFlips.delete(nodeId);
            if (this.flipQueue.size > 0) {
              this.processQueue();
            }
          }, 300); // Animation duration
        }, this.activeFlips.size * 50); // Stagger delay
      });

      this.animationFrame = null;
    });
  }

  registerFlip(nodeId: string, isFlipped: boolean) {
    if (isFlipped) {
      this.activeFlips.add(nodeId);
    } else {
      this.activeFlips.delete(nodeId);
    }

    // Emit metrics
    this.emitMetrics();
  }

  unflipAll() {
    this.activeFlips.forEach(nodeId => {
      const node = document.querySelector(`[data-node-id="${nodeId}"]`);
      node?.classList.remove('is-flipped');
    });

    this.activeFlips.clear();
    this.flipQueue.clear();
  }

  shouldBatch(): boolean {
    return this.activeFlips.size >= 5;
  }
}
```

### 5. Viewport Management

```typescript
// packages/core/hooks/useViewportManager.ts
export const useViewportManager = () => {
  const { getViewport, setViewport, getNodes, fitView } = useReactFlow();

  const adjustViewportForFlip = useCallback(
    async (nodeId: string) => {
      const node = getNodes().find(n => n.id === nodeId);
      if (!node) return;

      const viewport = getViewport();
      const nodeElement = document.querySelector(`[data-node-id="${nodeId}"]`);
      if (!nodeElement) return;

      const rect = nodeElement.getBoundingClientRect();
      const flippedWidth = rect.width * 1.1; // Account for shadow/padding
      const flippedHeight = rect.height * 1.1;

      // Check if flip would cause clipping
      const wouldClipRight = rect.right + flippedWidth > window.innerWidth;
      const wouldClipBottom = rect.bottom + flippedHeight > window.innerHeight;
      const wouldClipLeft = rect.left - flippedWidth < 0;
      const wouldClipTop = rect.top - flippedHeight < 0;

      if (wouldClipRight || wouldClipBottom || wouldClipLeft || wouldClipTop) {
        // Calculate adjustment
        const padding = 50;
        const targetX =
          node.position.x - (wouldClipLeft ? 100 : wouldClipRight ? -100 : 0);
        const targetY =
          node.position.y - (wouldClipTop ? 100 : wouldClipBottom ? -100 : 0);

        // Smooth pan to adjusted position
        const duration = 300;
        const steps = 30;
        const stepDuration = duration / steps;

        for (let i = 0; i <= steps; i++) {
          const progress = i / steps;
          const eased = easeInOutCubic(progress);

          const x = viewport.x + (targetX - viewport.x) * eased;
          const y = viewport.y + (targetY - viewport.y) * eased;

          setViewport({ x, y, zoom: viewport.zoom });

          await new Promise(resolve => setTimeout(resolve, stepDuration));
        }
      }
    },
    [getViewport, setViewport, getNodes]
  );

  return { adjustViewportForFlip };
};
```

### 6. Metadata Prefetching Service

```typescript
// packages/core/services/MetadataPrefetcher.ts
export class MetadataPrefetcher {
  private cache: Map<string, NodeMetadata> = new Map();
  private prefetchQueue: Set<string> = new Set();
  private idleCallback: number | null = null;

  constructor(private metadataService: MetadataService) {
    this.startIdlePrefetch();
  }

  private startIdlePrefetch() {
    if ('requestIdleCallback' in window) {
      this.idleCallback = requestIdleCallback(
        () => {
          this.processPrefetchQueue();
        },
        { timeout: 2000 }
      );
    } else {
      // Fallback for Safari
      setTimeout(() => this.processPrefetchQueue(), 1000);
    }
  }

  async prefetch(nodeIds: string[]) {
    nodeIds.forEach(id => {
      if (!this.cache.has(id)) {
        this.prefetchQueue.add(id);
      }
    });

    this.startIdlePrefetch();
  }

  private async processPrefetchQueue() {
    const batch = Array.from(this.prefetchQueue).slice(0, 5);

    await Promise.all(
      batch.map(async nodeId => {
        try {
          const metadata = await this.metadataService.getMetadata(nodeId);
          this.cache.set(nodeId, metadata);
          this.prefetchQueue.delete(nodeId);
        } catch (error) {
          console.warn(`Failed to prefetch metadata for ${nodeId}`, error);
        }
      })
    );

    if (this.prefetchQueue.size > 0) {
      this.startIdlePrefetch();
    }
  }

  get(nodeId: string): NodeMetadata | null {
    return this.cache.get(nodeId) || null;
  }

  invalidate(nodeId: string) {
    this.cache.delete(nodeId);
  }
}
```

```typescript
// packages/core/components/nodes/FlipAccessibility.tsx
export const FlipAccessibilityProvider: React.FC = ({ children }) => {
  useEffect(() => {
    const handleGlobalKeydown = (e: KeyboardEvent) => {
      // ESC to unflip current node
      if (e.key === 'Escape') {
        const focused = document.activeElement;
        const flippedNode = focused?.closest('.is-flipped');
        if (flippedNode) {
          flippedNode.classList.remove('is-flipped');
          announceToScreenReader('Returned to node view');
        }
      }

      // Shift+ESC to unflip all
      if (e.key === 'Escape' && e.shiftKey) {
        document.querySelectorAll('.is-flipped').forEach(node => {
          node.classList.remove('is-flipped');
        });
        announceToScreenReader('All nodes returned to normal view');
      }
    };

    document.addEventListener('keydown', handleGlobalKeydown);
    return () => document.removeEventListener('keydown', handleGlobalKeydown);
  }, []);

  return <>{children}</>;
};

function announceToScreenReader(message: string) {
  const announcement = document.createElement('div');
  announcement.setAttribute('role', 'status');
  announcement.setAttribute('aria-live', 'polite');
  announcement.className = 'sr-only';
  announcement.textContent = message;

  document.body.appendChild(announcement);
  setTimeout(() => document.body.removeChild(announcement), 1000);
}
```

```typescript
// packages/core/monitoring/FlipMetrics.ts
export class FlipPerformanceMonitor {
  private metrics: Map<string, FlipMetric> = new Map();

  recordFlip(nodeId: string, duration: number) {
    const metric: FlipMetric = {
      nodeId,
      duration,
      timestamp: Date.now(),
      viewportNodes: this.getVisibleNodeCount(),
      deviceType: this.getDeviceType(),
      reducedMotion: window.matchMedia('(prefers-reduced-motion: reduce)')
        .matches
    };

    this.metrics.set(nodeId, metric);

    // Report if performance degrades
    if (duration > 300) {
      this.reportSlowFlip(metric);
    }
  }

  getP95FlipTime(): number {
    const durations = Array.from(this.metrics.values())
      .map(m => m.duration)
      .sort((a, b) => a - b);

    const index = Math.floor(durations.length * 0.95);
    return durations[index] || 0;
  }

  private reportSlowFlip(metric: FlipMetric) {
    console.warn('Slow flip detected', {
      nodeId: metric.nodeId,
      duration: metric.duration,
      visibleNodes: metric.viewportNodes
    });

    // Send to analytics
    analytics.track('flip_performance_degradation', metric);
  }
}
```
