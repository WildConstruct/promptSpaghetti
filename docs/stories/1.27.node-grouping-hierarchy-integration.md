```typescript
// packages/core/utils/grouping.ts - UPDATED with caching

import { cache, workerPool, perfMonitor } from '../performance/infrastructure';

/**
 * Calculate group bounds with caching and performance monitoring
 * Leverages Story 0.1's multi-level cache for repeated calculations
 */
export async function getGroupBounds(
  group: NodeGroup,
  nodes: Node[]
): Promise<Bounds> {
  const cacheKey = `group:bounds:${group.id}:${group.nodeIds.join(',')}`;
  
  // Check cache first (L1 → L2 → L3)
  const cached = await cache.get(cacheKey);
  if (cached) {
    perfMonitor.recordCacheHit('groupBounds');
    return cached;
  }
  
  // For large groups, offload to worker
  if (group.nodeIds.length > 50) {
    const bounds = await perfMonitor.measure('groupBounds:worker', async () => {
      return workerPool.execute({
        type: 'CALCULATE_GROUP_BOUNDS',
        data: { group, nodes }
      });
    });
    
    // Cache the result
    await cache.set(cacheKey, bounds, { ttl: 60000 }); // 1 minute TTL
    return bounds;
  }
  
  // Small groups: calculate inline
  const bounds = perfMonitor.measure('groupBounds:inline', () => {
    const groupNodes = nodes.filter(n => group.nodeIds.includes(n.id));
    
    if (groupNodes.length === 0) {
      return { x: 0, y: 0, width: 0, height: 0 };
    }
    
    const xs = groupNodes.map(n => n.position.x);
    const ys = groupNodes.map(n => n.position.y);
    const rights = groupNodes.map(n => n.position.x + (n.width || 150));
    const bottoms = groupNodes.map(n => n.position.y + (n.height || 50));
    
    return {
      x: Math.min(...xs) - 20,
      y: Math.min(...ys) - 40,
      width: Math.max(...rights) - Math.min(...xs) + 40,
      height: Math.max(...bottoms) - Math.min(...ys) + 60
    };
  });
  
  // Cache the result
  await cache.set(cacheKey, bounds, { ttl: 60000 });
  return bounds;
}

/**
 * Optimized nested group operations using worker pool
 * Handles complex hierarchy calculations without blocking UI
 */
export async function validateGroupHierarchy(
  groups: Map<string, NodeGroup>,
  maxDepth: number = 3
): Promise<ValidationResult> {
  const cacheKey = `hierarchy:validation:${Array.from(groups.keys()).join(',')}`;
  
  // Try cache first
  const cached = await cache.get(cacheKey);
  if (cached) return cached;
  
  // Complex validation in worker to maintain 60fps
  const result = await workerPool.execute({
    type: 'VALIDATE_GROUP_HIERARCHY',
    data: {
      groups: Array.from(groups.entries()),
      maxDepth
    }
  });
  
  // Cache validation results
  await cache.set(cacheKey, result, { ttl: 30000 }); // 30 seconds
  return result;
}
```

### Group State Management with Performance

```typescript
// packages/core/stores/groupingSlice.ts - Performance-optimized store

interface GroupingSlice {
  groups: Map<string, NodeGroup>
  // Performance metrics from Story 0.1
  performanceMetrics: {
    lastOperationTime: number
    cacheHitRate: number
    workerUtilization: number
  }
  
  createGroup: (nodeIds: string[], name?: string) => Promise<void>
  deleteGroup: (groupId: string, deleteContents: boolean) => Promise<void>
  toggleGroup: (groupId: string) => Promise<void>
}

const createGroupingSlice = (set, get) => ({
  groups: new Map(),
  performanceMetrics: {
    lastOperationTime: 0,
    cacheHitRate: 0,
    workerUtilization: 0
  },
  
  createGroup: async (nodeIds: string[], name?: string) => {
    const startTime = performance.now();
    
    // Validate in worker for large selections
    if (nodeIds.length > 20) {
      const validation = await workerPool.execute({
        type: 'VALIDATE_GROUP_CREATION',
        data: { nodeIds, existingGroups: Array.from(get().groups.values()) }
      });
      
      if (!validation.valid) {
        throw new Error(validation.error);
      }
    }
    
    const group = createGroup(nodeIds, get().groups, name);
    
    // Update state
    set(state => ({
      groups: new Map(state.groups).set(group.id, group),
      performanceMetrics: {
        ...state.performanceMetrics,
        lastOperationTime: performance.now() - startTime
      }
    }));
    
    // Invalidate affected caches
    await cache.invalidatePattern(`group:bounds:*`);
    
    // Track performance
    perfMonitor.record('createGroup', performance.now() - startTime);
  },
  
  toggleGroup: async (groupId: string) => {
    const startTime = performance.now();
    const group = get().groups.get(groupId);
    if (!group) return;
    
    // For large groups, calculate collapse in worker
    if (group.nodeIds.length > 50) {
      const collapsedData = await workerPool.execute({
        type: 'CALCULATE_COLLAPSED_GROUP',
        data: { group, nodes: get().nodes }
      });
      
      group.collapsedPosition = collapsedData.position;
      group.collapsedSize = collapsedData.size;
    }
    
    group.collapsed = !group.collapsed;
    
    set(state => ({
      groups: new Map(state.groups).set(groupId, group),
      performanceMetrics: {
        ...state.performanceMetrics,
        lastOperationTime: performance.now() - startTime
      }
    }));
    
    perfMonitor.record('toggleGroup', performance.now() - startTime);
  }
});
```

```typescript
// packages/core/hooks/useGroupedNodes.ts - Performance-optimized hook

export function useGroupedNodes(nodes: Node[], groups: Map<string, NodeGroup>) {
  const [cachedResult, setCachedResult] = useState<Node[]>([]);
  const [isCalculating, setIsCalculating] = useState(false);
  
  useEffect(() => {
    const calculateGroupedNodes = async () => {
      const cacheKey = `grouped:nodes:${nodes.length}:${groups.size}`;
      
      // Try cache first
      const cached = await cache.get(cacheKey);
      if (cached) {
        setCachedResult(cached);
        return;
      }
      
      setIsCalculating(true);
      
      // For large graphs, use worker
      if (nodes.length > 100 || groups.size > 20) {
        const result = await workerPool.execute({
          type: 'CALCULATE_GROUPED_NODES',
          data: {
            nodes,
            groups: Array.from(groups.entries())
          }
        });
        
        await cache.set(cacheKey, result, { ttl: 5000 }); // 5 second TTL
        setCachedResult(result);
      } else {
        // Small graphs: calculate inline
        const visibleNodes: Node[] = [];
        const hiddenNodeIds = new Set<string>();
        
        groups.forEach(group => {
          if (group.collapsed) {
            group.nodeIds.forEach(id => hiddenNodeIds.add(id));
            visibleNodes.push(collapseGroup(group, nodes));
          }
        });
        
        nodes.forEach(node => {
          if (!hiddenNodeIds.has(node.id)) {
            visibleNodes.push(node);
          }
        });
        
        await cache.set(cacheKey, visibleNodes, { ttl: 5000 });
        setCachedResult(visibleNodes);
      }
      
      setIsCalculating(false);
    };
    
    calculateGroupedNodes();
  }, [nodes, groups]);
  
  return { groupedNodes: cachedResult, isCalculating };
}
```
