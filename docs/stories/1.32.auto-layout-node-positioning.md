# Story 1.32: Auto-Layout and Node Positioning System

## Story ID
STORY-1.32

## Title
Implement Auto-Layout System to Prevent Node Stacking and Enable Graph Cleanup

## Priority
High

## Story Points
5

## Description
As a user, I want nodes dragged from the asset browser to automatically arrange themselves without overlapping, and I want a "clean up" command to organize messy node layouts, so that my graphs remain readable and well-organized.

## Problem Statement
Currently, when dragging assets with multiple nodes from the asset browser, all nodes end up stacked on top of each other at the drop position. Users must manually separate and arrange them, which is tedious and time-consuming. Additionally, graphs can become messy over time with no easy way to reorganize them.

## Acceptance Criteria
1. ✅ When dropping an asset with multiple nodes, nodes automatically arrange with proper spacing
2. ✅ Nodes never overlap when initially placed from asset browser
3. ✅ "Clean up" command available via keyboard shortcut (Cmd+Shift+L)
4. ✅ "Clean up" command available in context menu for selected nodes
5. ✅ Clean up works on: entire graph, selected nodes only, or newly dropped nodes
6. ✅ Layout preserves logical flow direction (inputs → outputs)
7. ✅ Animation smoothly transitions nodes to new positions
8. ✅ Undo/redo support for layout operations

## Technical Specifications

### Layout Algorithms

#### 1. Dagre (Hierarchical Layout) - PRIMARY
```typescript
import dagre from 'dagre';

export function applyDagreLayout(nodes: Node[], edges: Edge[]): Node[] {
  const g = new dagre.graphlib.Graph();
  g.setGraph({ 
    rankdir: 'LR', // Left to right
    nodesep: 100,  // Horizontal spacing
    ranksep: 150,  // Vertical spacing
    marginx: 20,
    marginy: 20
  });
  
  // Add nodes and edges to graph
  nodes.forEach(node => {
    g.setNode(node.id, { 
      width: node.width || 200, 
      height: node.height || 80 
    });
  });
  
  edges.forEach(edge => {
    g.setEdge(edge.source, edge.target);
  });
  
  // Calculate layout
  dagre.layout(g);
  
  // Apply positions
  return nodes.map(node => {
    const position = g.node(node.id);
    return {
      ...node,
      position: {
        x: position.x - (node.width || 200) / 2,
        y: position.y - (node.height || 80) / 2
      }
    };
  });
}
```

#### 2. Force-Directed Layout (Spring Layout) - SECONDARY
```typescript
import * as d3 from 'd3-force';

export function applyForceLayout(nodes: Node[], edges: Edge[]): Node[] {
  const simulation = d3.forceSimulation(nodes)
    .force('link', d3.forceLink(edges)
      .id((d: any) => d.id)
      .distance(150))
    .force('charge', d3.forceManyBody()
      .strength(-500))
    .force('center', d3.forceCenter(400, 300))
    .force('collision', d3.forceCollide()
      .radius(100));
  
  // Run simulation
  simulation.tick(300);
  
  return nodes.map(node => ({
    ...node,
    position: {
      x: (node as any).x,
      y: (node as any).y
    }
  }));
}
```

#### 3. Grid-Based Layout (Simple) - FALLBACK
```typescript
export function applyGridLayout(nodes: Node[], startPos: { x: number, y: number }): Node[] {
  const GRID_SIZE = 250;
  const COLUMNS = Math.ceil(Math.sqrt(nodes.length));
  
  return nodes.map((node, index) => {
    const row = Math.floor(index / COLUMNS);
    const col = index % COLUMNS;
    
    return {
      ...node,
      position: {
        x: startPos.x + col * GRID_SIZE,
        y: startPos.y + row * GRID_SIZE
      }
    };
  });
}
```

### Auto-Layout Hook
```typescript
import { useCallback } from 'react';
import { Node, Edge, useReactFlow } from 'reactflow';

export function useAutoLayout() {
  const { setNodes, getNodes, getEdges, fitView } = useReactFlow();
  
  const cleanupNodes = useCallback((
    nodesToClean?: Node[],
    algorithm: 'dagre' | 'force' | 'grid' = 'dagre'
  ) => {
    const nodes = nodesToClean || getNodes();
    const edges = getEdges();
    
    let layoutedNodes: Node[];
    
    switch (algorithm) {
      case 'dagre':
        layoutedNodes = applyDagreLayout(nodes, edges);
        break;
      case 'force':
        layoutedNodes = applyForceLayout(nodes, edges);
        break;
      case 'grid':
        layoutedNodes = applyGridLayout(nodes, { x: 0, y: 0 });
        break;
    }
    
    // Animate to new positions
    setNodes((allNodes) => {
      return allNodes.map((node) => {
        const layouted = layoutedNodes.find(n => n.id === node.id);
        if (layouted) {
          return {
            ...node,
            position: layouted.position,
            style: {
              ...node.style,
              transition: 'all 0.5s ease'
            }
          };
        }
        return node;
      });
    });
    
    // Fit view after animation
    setTimeout(() => fitView({ padding: 0.1 }), 600);
  }, [setNodes, getNodes, getEdges, fitView]);
  
  return { cleanupNodes };
}
```

### Integration with Drag & Drop
```typescript
// In useDragDropHandlers.ts
const handleDrop = useCallback((event) => {
  const preset = JSON.parse(event.dataTransfer.getData('application/json'));
  const dropPosition = reactFlowInstance.project({ 
    x: event.clientX, 
    y: event.clientY 
  });
  
  // Create nodes from preset
  const newNodes = createNodesFromPreset(preset, dropPosition);
  
  // Auto-layout if multiple nodes
  if (newNodes.length > 1) {
    const layoutedNodes = applyDagreLayout(newNodes, []);
    setNodes((nodes) => [...nodes, ...layoutedNodes]);
  } else {
    setNodes((nodes) => [...nodes, ...newNodes]);
  }
  
  showNotification('success', `Added ${newNodes.length} nodes`);
}, [reactFlowInstance, setNodes]);
```

```typescript
// In useKeyboardHandlers.ts
useEffect(() => {
  const handleKeyDown = (e: KeyboardEvent) => {
    // Cmd+Shift+L for cleanup
    if (e.metaKey && e.shiftKey && e.key === 'l') {
      e.preventDefault();
      const selectedNodes = nodes.filter(n => n.selected);
      cleanupNodes(selectedNodes.length > 0 ? selectedNodes : undefined);
      showNotification('info', 'Layout cleaned up');
    }
  };
  
  window.addEventListener('keydown', handleKeyDown);
  return () => window.removeEventListener('keydown', handleKeyDown);
}, [nodes, cleanupNodes]);
```

## Performance Considerations
- **Debounce**: Prevent rapid re-layouts during drag operations
- **Batch updates**: Apply all position changes in single setState
- **Web Workers**: Consider moving complex calculations to worker thread for large graphs (100+ nodes)
- **Incremental layout**: Only re-layout affected subgraph when possible

## Edge Cases
1. **Disconnected components**: Handle multiple separate subgraphs
2. **Circular dependencies**: Ensure algorithm handles cycles gracefully
3. **Fixed nodes**: Respect manually positioned/locked nodes
4. **Grouped nodes**: Maintain group relationships during layout
5. **Very large graphs**: Provide progress indicator for 100+ nodes
