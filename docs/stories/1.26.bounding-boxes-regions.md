**Geometry Utilities:**
```typescript
// Check if node is within box bounds
export function isNodeInBox(
  node: Node,
  bounds: Bounds
): boolean {
  const nodeRect = {
    x: node.position.x,
    y: node.position.y,
    width: node.width || 150,
    height: node.height || 50
  }
  
  return (
    nodeRect.x >= bounds.x &&
    nodeRect.y >= bounds.y &&
    nodeRect.x + nodeRect.width <= bounds.x + bounds.width &&
    nodeRect.y + nodeRect.height <= bounds.y + bounds.height
  )
}

// Get all nodes within box
export function getNodesInBox(
  nodes: Node[],
  bounds: Bounds
): Node[] {
  return nodes.filter(node => isNodeInBox(node, bounds))
}
```
