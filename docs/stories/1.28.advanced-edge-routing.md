**Routing Algorithms:**
```typescript
// Bezier curve with control points
export function calculateBezierPath(
  x1: number, y1: number,
  x2: number, y2: number,
  controlPoints: ControlPoint[]
): string {
  if (controlPoints.length === 0) {
    // Simple quadratic bezier
    const cx = (x1 + x2) / 2
    const cy = (y1 + y2) / 2
    return `M ${x1},${y1} Q ${cx},${cy} ${x2},${y2}`
  }
  
  // Build path through control points
  let path = `M ${x1},${y1}`
  const points = [
    { x: x1, y: y1 },
    ...controlPoints,
    { x: x2, y: y2 }
  ]
  
  for (let i = 1; i < points.length; i++) {
    const prev = points[i - 1]
    const curr = points[i]
    const next = points[i + 1]
    
    if (next) {
      // Cubic bezier through point
      const cp1x = prev.x + (curr.x - prev.x) * 0.5
      const cp1y = prev.y + (curr.y - prev.y) * 0.5
      const cp2x = curr.x + (next.x - curr.x) * 0.5
      const cp2y = curr.y + (next.y - curr.y) * 0.5
      
      path += ` C ${cp1x},${cp1y} ${cp2x},${cp2y} ${curr.x},${curr.y}`
    } else {
      // Last segment
      path += ` L ${curr.x},${curr.y}`
    }
  }
  
  return path
}

// Orthogonal routing (Manhattan routing)
export function calculateOrthogonalPath(
  x1: number, y1: number,
  x2: number, y2: number,
  controlPoints: ControlPoint[]
): string {
  const segments: string[] = [`M ${x1},${y1}`]
  
  if (controlPoints.length === 0) {
    // Auto-calculate path
    const midX = (x1 + x2) / 2
    segments.push(`L ${midX},${y1}`) // Horizontal
    segments.push(`L ${midX},${y2}`) // Vertical
    segments.push(`L ${x2},${y2}`)   // Horizontal
  } else {
    // Use control points as corners
    let currentX = x1
    let currentY = y1
    
    controlPoints.forEach((point, i) => {
      // Alternate between horizontal and vertical
      if (i % 2 === 0) {
        segments.push(`L ${point.x},${currentY}`) // Horizontal
        segments.push(`L ${point.x},${point.y}`)  // Vertical
      } else {
        segments.push(`L ${currentX},${point.y}`) // Vertical
        segments.push(`L ${point.x},${point.y}`)  // Horizontal
      }
      currentX = point.x
      currentY = point.y
    })
    
    // Connect to target
    segments.push(`L ${x2},${currentY}`)
    segments.push(`L ${x2},${y2}`)
  }
  
  return segments.join(' ')
}

// Auto-routing to avoid obstacles
export function autoRouteEdges(
  edges: Edge[],
  nodes: Node[]
): Edge[] {
  const obstacles = nodes.map(node => ({
    x: node.position.x,
    y: node.position.y,
    width: node.width || 150,
    height: node.height || 50
  }))
  
  return edges.map(edge => {
    const source = nodes.find(n => n.id === edge.source)
    const target = nodes.find(n => n.id === edge.target)
    
    if (!source || !target) return edge
    
    // Find path avoiding obstacles
    const path = findPath(
      source.position,
      target.position,
      obstacles
    )
    
    // Convert path to control points
    const controlPoints = path
      .slice(1, -1) // Exclude start and end
      .map((point, i) => ({
        id: `cp-${i}`,
        x: point.x,
        y: point.y,
        type: 'sharp' as const
      }))
    
    return {
      ...edge,
      data: {
        ...edge.data,
        controlPoints,
        routingStyle: 'orthogonal'
      }
    }
  })
}
```

##### Critical Performance Concerns:

1. **Path Calculation Overhead**:
   ```typescript
   // Current approach recalculates on every render
   // Better: Cache paths and invalidate selectively
   const pathCache = new Map<string, string>();
   
   const getCachedPath = (key: string, calculator: () => string) => {
     if (!pathCache.has(key)) {
       pathCache.set(key, calculator());
     }
     return pathCache.get(key)!;
   };
   ```

2. **Auto-Routing Complexity**:
   - O(n²) for collision detection with naive approach
   - Need spatial partitioning for large graphs
   - Consider web worker for heavy computation

   - Control points add significant memory per edge
   - Need cleanup for deleted edges
   - Consider virtualization for off-screen edges
