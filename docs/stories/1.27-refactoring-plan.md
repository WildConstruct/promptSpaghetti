#### useNodeContainmentCache Hook
```typescript
// hooks/useNodeContainment.ts
import { PerformanceMonitor } from '@/packages/core/utils/performance/PerformanceMonitor';

export function useNodeContainmentCache(
  boxId: string,
  nodes: Node[],
  boxPosition: Position,
  boxSize: Size,
  isExpanded: boolean
) {
  const perfMonitor = PerformanceMonitor.getInstance();
  const cacheRef = useRef<Map<string, Node[]>>(new Map());
  
  return useMemo(() => {
    // Use expanded size for containment check even when collapsed
    const checkSize = isExpanded ? boxSize : expandedSizeRef.current;
    const cacheKey = `${boxId}:${nodes.length}:${checkSize.width}x${checkSize.height}`;
    
    if (cacheRef.current.has(cacheKey)) {
      perfMonitor.record('boundingBox.cacheHit', 1);
      return cacheRef.current.get(cacheKey)!;
    }
    
    const start = performance.now();
    const contained = calculateContainedNodes(nodes, boxId, boxPosition, checkSize);
    
    perfMonitor.record('boundingBox.containmentCalc', performance.now() - start);
    cacheRef.current.set(cacheKey, contained);
    
    // LRU cache eviction
    if (cacheRef.current.size > CACHE_SIZE) {
      const firstKey = cacheRef.current.keys().next().value;
      cacheRef.current.delete(firstKey);
    }
    
    return contained;
  }, [boxId, nodes.length, boxPosition.x, boxPosition.y, boxSize, isExpanded]);
}
```

#### useCollapseAnimation Hook (RAF-based)
```typescript
// hooks/useCollapseAnimation.ts
export function useCollapseAnimation(
  isCollapsed: boolean,
  expandedSize: Size,
  onAnimationComplete: () => void
) {
  const [size, setSize] = useState<Size>(
    isCollapsed ? COLLAPSED_SIZE : expandedSize
  );
  const rafRef = useRef<number>();
  const startTimeRef = useRef<number>();
  
  const animate = useCallback((timestamp: number) => {
    if (!startTimeRef.current) startTimeRef.current = timestamp;
    
    const progress = Math.min(
      (timestamp - startTimeRef.current) / COLLAPSE_DURATION,
      1
    );
    
    // Cubic ease-in-out
    const eased = progress < 0.5
      ? 4 * progress * progress * progress
      : 1 - Math.pow(-2 * progress + 2, 3) / 2;
    
    const currentSize = interpolateSize(
      isCollapsed ? expandedSize : COLLAPSED_SIZE,
      isCollapsed ? COLLAPSED_SIZE : expandedSize,
      eased
    );
    
    setSize(currentSize);
    
    if (progress < 1) {
      rafRef.current = requestAnimationFrame(animate);
    } else {
      onAnimationComplete();
    }
  }, [isCollapsed, expandedSize, onAnimationComplete]);
  
  useEffect(() => {
    startTimeRef.current = undefined;
    rafRef.current = requestAnimationFrame(animate);
    
    return () => {
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
    };
  }, [isCollapsed]);
  
  return { size, isAnimating: !!rafRef.current };
}
```

```typescript
// ResizeHandles.tsx
const RESIZE_DIRECTIONS = ['n', 'ne', 'e', 'se', 's', 'sw', 'w', 'nw'] as const;

export const ResizeHandles: React.FC<ResizeHandlesProps> = React.memo(({
  visible,
  isLocked,
  onResizeStart,
}) => {
  if (!visible || isLocked) return null;
  
  return (
    <>
      {RESIZE_DIRECTIONS.map(direction => (
        <ResizeHandle
          key={direction}
          direction={direction}
          onMouseDown={(e) => onResizeStart(e, direction)}
        />
      ))}
    </>
  );
});

const ResizeHandle: React.FC<HandleProps> = React.memo(({
  direction,
  onMouseDown,
}) => {
  const style = getResizeHandleStyle(direction);
  
  return (
    <div
      className="nodrag resize-handle"
      style={style}
      onMouseDown={(e) => {
        e.stopPropagation();
        e.preventDefault();
        onMouseDown(e);
      }}
      onPointerDown={(e) => e.stopPropagation()}
    />
  );
});
```

### Phase 4: Performance Optimizations

#### Key Optimizations
1. **Memoization**: All expensive calculations cached
2. **RAF Animations**: Smooth 60fps animations
3. **Debounced Resize**: Reduce state updates during drag
4. **Worker Offloading**: Large node calculations in worker
5. **Virtual Rendering**: Only render visible nodes when expanded
