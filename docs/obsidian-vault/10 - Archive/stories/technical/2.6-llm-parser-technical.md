### 3. Security Layer Implementation

```typescript
// packages/core/services/ParserSecurity.ts
export class ParserSecurity {
  private readonly PII_PATTERNS = {
    email: /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/gi,
    phone: /\b\d{3}[-.]?\d{3}[-.]??\d{4}\b/g,
    ssn: /\b\d{3}-\d{2}-\d{4}\b/g,
    creditCard: /\b(?:\d{4}[-\s]?){3}\d{4}\b/g,
    ipAddress: /\b(?:\d{1,3}\.){3}\d{1,3}\b/g
  };

  private readonly INJECTION_PATTERNS = [
    /ignore (previous|all) instructions/gi,
    /system:/gi,
    /\[INST\]/gi,
    /<\|im_start\|>/gi,
    /\{\{.*\}\}/g, // Jinja templates
    /<script.*?>/gi,
    /javascript:/gi
  ];

  sanitizeForLLM(prompt: string, options: SanitizeOptions): SanitizedResult {
    let sanitized = prompt;
    const redactions: Redaction[] = [];

    if (options.maskPII) {
      Object.entries(this.PII_PATTERNS).forEach(([type, pattern]) => {
        sanitized = sanitized.replace(pattern, (match, offset) => {
          const token = `[PII_${type.toUpperCase()}_${offset}]`;
          redactions.push({ type, original: match, token, offset });
          return token;
        });
      });
    }

    // Remove injection attempts
    this.INJECTION_PATTERNS.forEach(pattern => {
      sanitized = sanitized.replace(pattern, '');
    });

    return { sanitized, redactions, hash: this.hashContent(sanitized) };
  }

  validateLLMResponse(response: unknown): SecurityValidation {
    const str = JSON.stringify(response);
    const dangerous = [
      'eval(',
      'Function(',
      '__proto__',
      'constructor',
      'require(',
      'import(',
      'fetch(',
      'XMLHttpRequest'
    ];

    const violations = dangerous.filter(pattern => str.includes(pattern));

    return {
      safe: violations.length === 0,
      violations,
      sanitized:
        violations.length > 0 ? this.stripDangerous(response) : response
    };
  }
}
```

### 4. Performance & Caching Layer

```typescript
// packages/core/services/ParserCache.ts
export class ParserCache {
  private cache: LRUCache<string, CachedParse>;
  private metrics: CacheMetrics;

  async get(prompt: string, mode: ParserMode): Promise<ParseResult | null> {
    const key = this.generateKey(prompt, mode);
    const cached = this.cache.get(key);

    if (cached && !this.isStale(cached)) {
      this.metrics.hit();
      return cached.result;
    }

    this.metrics.miss();
    return null;
  }

  private generateKey(prompt: string, mode: ParserMode): string {
    const normalized = prompt.toLowerCase().trim();
    const contentHash = crypto
      .createHash('sha256')
      .update(normalized)
      .digest('hex')
      .substring(0, 16);

    return `${mode}_${contentHash}_v${PARSER_VERSION}`;
  }
}

// packages/core/services/ParserMetrics.ts
export class ParserMetrics {
  private metrics: Map<string, Metric[]> = new Map();

  recordParse(mode: ParserMode, duration: number, result: ParseResult) {
    const metrics = {
      parse_latency_ms: duration,
      node_count: result.nodes.length,
      edge_count: result.edges.length,
      cache_hit: result.fromCache ? 1 : 0,
      mode,
      timestamp: Date.now()
    };

    this.emit('parse_complete', metrics);

    if (mode === 'llm-enhanced') {
      this.recordLLMMetrics(result);
    }
  }

  getMetrics(): AggregatedMetrics {
    return {
      parse_latency_p50: this.percentile('parse_latency_ms', 50),
      parse_latency_p95: this.percentile('parse_latency_ms', 95),
      cache_hit_rate: this.average('cache_hit'),
      fallback_rate: this.calculateFallbackRate(),
      node_f1: this.latestEvalScore('node_f1'),
      edge_f1: this.latestEvalScore('edge_f1'),
      variable_integrity: this.latestEvalScore('variable_integrity')
    };
  }
}
```

### 6. Fallback & Error Handling

```typescript
// packages/core/services/ParserFallback.ts
export class ParserFallback {
  async executeParse(
    prompt: string,
    options: ParserOptions
  ): Promise<ParseResult> {
    const startTime = performance.now();

    try {
      // Try LLM with retries
      const result = await this.tryLLMParse(prompt, options);

      // Validate result quality
      if (this.isQualityAcceptable(result)) {
        return result;
      }

      throw new Error('LLM parse quality below threshold');
    } catch (error) {
      // Log for monitoring
      logger.warn('LLM parse failed, falling back', {
        error: error.message,
        duration: performance.now() - startTime,
        promptLength: prompt.length
      });

      // Show non-blocking toast
      toast.info('Using standard parser (LLM unavailable)', {
        action: {
          label: 'Details',
          onClick: () => showErrorDetails(error)
        }
      });

      // Fallback to standard with enhanced heuristics
      return this.enhancedStandardParse(prompt, options);
    }
  }

  private async tryLLMParse(
    prompt: string,
    options: ParserOptions
  ): Promise<ParseResult> {
    const retryDelays = [300, 800]; // Exponential backoff
    let lastError: Error;

    for (let attempt = 0; attempt <= retryDelays.length; attempt++) {
      try {
        const timeout = new Promise((_, reject) =>
          setTimeout(() => reject(new Error('Timeout')), 3000)
        );

        const parse = this.llmService.parse(prompt, options);

        return (await Promise.race([parse, timeout])) as ParseResult;
      } catch (error) {
        lastError = error;

        if (attempt < retryDelays.length) {
          await new Promise(resolve =>
            setTimeout(resolve, retryDelays[attempt])
          );
        }
      }
    }

    throw lastError!;
  }
}
```

```typescript
// packages/eval/scripts/ci-integration.ts
export class ParserEvaluator {
  private goldenTests: GoldenTest[];

  async evaluate(parser: PromptParser): Promise<EvalResults> {
    const results: Score[] = [];

    for (const test of this.goldenTests) {
      const parsed = await parser.parse(test.prompt, {
        mode: 'llm-enhanced',
        preserveVariables: true,
        autoConnect: true
      });

      const score = scoreGraph(test.truth, parsed);
      results.push(score);

      // Early exit if failing
      if (
        score.nodeF1 < 0.7 ||
        score.edgeF1 < 0.6 ||
        score.variableIntegrity < 0.995
      ) {
        console.error(`❌ Test ${test.id} failed quality thresholds`);
        console.error(`  Node F1: ${score.nodeF1} (min: 0.7)`);
        console.error(`  Edge F1: ${score.edgeF1} (min: 0.6)`);
        console.error(
          `  Variable Integrity: ${score.variableIntegrity} (min: 0.995)`
        );
      }
    }

    return this.aggregateResults(results);
  }
}
```
