### Performance Monitoring

```typescript
// PerformanceMonitor.ts
export class PerformanceMonitor {
  private metrics: Map<string, Metric[]> = new Map();

  track(operation: string, duration: number, metadata?: any) {
    this.metrics.get(operation)?.push({
      duration,
      timestamp: Date.now(),
      ...metadata
    });
  }

  getP95(operation: string): number {
    const values = this.metrics.get(operation) || [];
    return calculatePercentile(values, 95);
  }

  generateReport(): PerformanceReport {
    return {
      p95Latency: this.getP95('llm_call'),
      cacheHitRate: this.calculateCacheRate(),
      costPerOperation: this.calculateCosts(),
      recommendations: this.analyzeBottlenecks()
    };
  }
}
```
